  // tick simulated clock forward from 08:45
  const clk = document.getElementById('clock');
  const start = Date.now();
  function pad(n){return String(n).padStart(2,'0')}
  function tick(){
    const el = (Date.now()-start)/1000;
    const baseSec = 8*3600 + 45*60;
    const total = baseSec + Math.floor(el);
    const h = Math.floor(total/3600)%24, m = Math.floor(total/60)%60, s = total%60;
    clk.textContent = `${pad(h)}:${pad(m)}:${pad(s)}`;
  }
  setInterval(tick, 1000);

  function tick(){
    const el = (Date.now()-start)/1000;
    const baseSec = 9*3600 + 50*60;
    const total = baseSec + Math.floor(el);
    const h = Math.floor(total/3600)%24, m = Math.floor(total/60)%60, s = total%60;

  }
